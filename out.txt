/// Reloj de animación compartido por todos los fractales.
/// `velocidad` = elementos (segmentos, nodos, niveles...) por segundo.
pub struct Animacion {
    pub animando: bool,
    pub velocidad: f32,
    inicio: Option<f64>,
}

impl Animacion {
    pub fn new(velocidad: f32) -> Self {
        Self { animando: true, velocidad, inicio: None }
    }

    /// Reinicia la animación. El reloj arranca en el siguiente frame
    /// (así no se "salta" la animación si la app lleva rato abierta).
    pub fn reiniciar(&mut self) {
        self.animando = true;
        self.inicio = None;
    }

    /// Devuelve cuántos elementos (de `total`) se deben mostrar ahora.
    /// Pide repintado solo mientras la animación sigue en curso.
    pub fn progreso(&mut self, ui: &egui::Ui, total: usize) -> usize {
        if !self.animando {
            return total;
        }
        let ahora = ui.input(|i| i.time);
        let inicio = *self.inicio.get_or_insert(ahora);
        let n = ((ahora - inicio).max(0.0) * self.velocidad as f64) as usize;
        if n >= total {
            self.animando = false;
            total
        } else {
            ui.ctx().request_repaint();
            n
        }
    }
}
use eframe::egui;

pub struct Camara {
    pub zoom: f32,
    pub offset: egui::Vec2,
}

impl Default for Camara {
    fn default() -> Self {
        Self { zoom: 1.0, offset: egui::Vec2::ZERO }
    }
}

impl Camara {
    pub fn interactuar(&mut self, ui: &mut egui::Ui, rect: egui::Rect) {
        let resp = ui.allocate_rect(rect, egui::Sense::drag());

        if resp.dragged() {
            self.offset += resp.drag_delta();
        }

        if resp.hovered() {
            let scroll = ui.input(|i| i.smooth_scroll_delta.y);
            if scroll != 0.0 {
                if let Some(cursor) = resp.hover_pos() {
                    let centro = rect.center();
                    let zoom_viejo = self.zoom;
                    let zoom_nuevo = (zoom_viejo * (scroll * 0.001).exp()).clamp(0.01, 100.0);
                    let mundo = (cursor - centro - self.offset) / zoom_viejo;
                    self.offset = cursor - centro - mundo * zoom_nuevo;
                    self.zoom = zoom_nuevo;
                }
            }
        }
    }

    pub fn origen(&self, centro: egui::Pos2) -> egui::Pos2 {
        centro + self.offset
    }

    pub fn reset(&mut self) {
        self.zoom = 1.0;
        self.offset = egui::Vec2::ZERO;
    }
}
//! Curvas (polilíneas) cuyo trabajo se reparte entre "hilos".
//! Secuencial = 1 trabajador con toda la curva; paralelo = bloques contiguos, uno por trabajador.

use crate::camara::Camara;
use crate::dibujo;
use crate::hilos::FilaHilo;
use eframe::egui::{Color32, Painter, Pos2, Stroke};

struct Tramo {
    ini: usize,
    fin: usize,
    hilo: usize,
}

pub struct Curva {
    /// Vértices en coordenadas del mundo (segmentos + 1).
    puntos: Vec<Pos2>,
    tramos: Vec<Tramo>,
    trabajo: Vec<usize>,
    /// Segmentos del trabajador con más carga (define la duración de la animación).
    pub max_trabajo: usize,
}

impl Curva {
    pub fn vacia() -> Self {
        Self::secuencial(Vec::new())
    }

    pub fn secuencial(puntos: Vec<Pos2>) -> Self {
        let n = puntos.len().saturating_sub(1);
        Self {
            puntos,
            tramos: vec![Tramo { ini: 0, fin: n, hilo: 0 }],
            trabajo: vec![n],
            max_trabajo: n,
        }
    }

    /// Divide la curva en `w` bloques contiguos del mismo tamaño.
    pub fn bloques(puntos: Vec<Pos2>, w: usize) -> Self {
        let n = puntos.len().saturating_sub(1);
        let w = w.max(1);
        let tam = ((n + w - 1) / w).max(1);
        let mut tramos = Vec::new();
        let mut trabajo = Vec::new();
        for b in 0..w {
            let ini = b * tam;
            if ini >= n {
                break;
            }
            let fin = ((b + 1) * tam).min(n);
            tramos.push(Tramo { ini, fin, hilo: b });
            trabajo.push(fin - ini);
        }
        let max_trabajo = trabajo.iter().copied().max().unwrap_or(0);
        Self { puntos, tramos, trabajo, max_trabajo }
    }

    /// `n` = segmentos que ya dibujó cada trabajador. Mientras un trabajador sigue
    /// dibujando se marca su posición actual con un punto del color de su hilo.
    pub fn dibujar(
        &self,
        painter: &Painter,
        camara: &Camara,
        centro: Pos2,
        n: usize,
        animando: bool,
        paso: usize,
        grosor: f32,
        color: impl Fn(usize) -> Color32,
    ) {
        for t in &self.tramos {
            let largo = t.fin - t.ini;
            let k = n.min(largo);
            if k == 0 {
                continue;
            }
            let c = color(t.hilo);
            dibujo::polilinea(
                painter,
                camara,
                centro,
                &self.puntos[t.ini..=t.ini + k],
                paso,
                Stroke::new(grosor, c),
            );

            if animando && k < largo {
                let cabeza = camara.origen(centro) + self.puntos[t.ini + k].to_vec2() * camara.zoom;
                painter.circle_filled(cabeza, 3.5, c);
            }
        }
    }

    pub fn filas(&self, n: usize) -> Vec<FilaHilo> {
        self.trabajo
            .iter()
            .enumerate()
            .map(|(h, &t)| FilaHilo {
                hilo: h,
                progreso: if t == 0 { 1.0 } else { (n as f32 / t as f32).min(1.0) },
            })
            .collect()
    }
}
//! Rutinas de dibujo por lotes: pocas `Shape` grandes en vez de miles pequeñas,
//! con culling, nivel de detalle y un tope de vértices por frame.

use crate::camara::Camara;
use crate::hilos;
use eframe::egui::{self, Color32, Mesh, Pos2, Shape, Stroke, Vec2};
use std::sync::OnceLock;

/// Puntos por trozo de polilínea. Al partirla, egui descarta los trozos
/// que quedan completamente fuera del panel.
const TROZO: usize = 2048;
const MAX_PUNTOS_POLILINEA: usize = 400_000;
const MAX_BORDES: usize = 20_000;
pub const MAX_VERTICES_MESH: usize = 60_000;
/// Tope de vértices por frame. El buffer de egui_wgpu crece a potencias de 2:
/// por encima de ~8,4 M vértices pide más de 256 MiB y wgpu aborta.
pub const MAX_VERTICES_FRAME: usize = 3_000_000;

#[derive(Clone, Copy)]
pub struct Cuadrado {
    pub p: [Pos2; 4],
    pub lado: f32,
    pub color: Color32,
    pub nivel: u8,
    pub hilo: u8,
    /// Instante (en unidades de trabajo) en que su hilo lo termina.
    pub tiempo: f32,
}

/// Acumula triángulos en `Mesh`es de tamaño acotado.
pub struct Lote {
    pub mesh: Mesh,
    shapes: Vec<Shape>,
    volcados: usize,
}

impl Lote {
    pub fn nuevo() -> Self {
        Self { mesh: Mesh::default(), shapes: Vec::new(), volcados: 0 }
    }

    /// Llamar tras añadir geometría: parte el mesh cuando se hace grande.
    pub fn revisar(&mut self) {
        if self.mesh.vertices.len() >= MAX_VERTICES_MESH {
            self.volcados += self.mesh.vertices.len();
            self.shapes.push(Shape::mesh(std::mem::take(&mut self.mesh)));
        }
    }

    pub fn lleno(&self) -> bool {
        self.volcados + self.mesh.vertices.len() >= MAX_VERTICES_FRAME
    }

    pub fn pintar(mut self, painter: &egui::Painter) {
        if !self.mesh.is_empty() {
            self.shapes.push(Shape::mesh(self.mesh));
        }
        painter.extend(self.shapes);
    }
}

/// Dibuja una polilínea dada en coordenadas del mundo.
/// `paso` > 1 salta puntos (nivel de detalle) cuando los segmentos miden < 1 px.
pub fn polilinea(
    painter: &egui::Painter,
    camara: &Camara,
    centro: Pos2,
    puntos: &[Pos2],
    paso: usize,
    stroke: Stroke,
) {
    if puntos.len() < 2 {
        return;
    }
    let mut paso = paso.max(1);
    if puntos.len() / paso > MAX_PUNTOS_POLILINEA {
        paso = puntos.len() / MAX_PUNTOS_POLILINEA + 1;
    }
    let origen = camara.origen(centro);
    let z = camara.zoom;
    let ultimo = puntos.len() - 1;

    let mut shapes: Vec<Shape> = Vec::new();
    let mut buf: Vec<Pos2> = Vec::with_capacity(TROZO + 1);
    let mut i = 0usize;
    loop {
        let s = origen + puntos[i].to_vec2() * z;
        buf.push(s);
        if buf.len() > TROZO {
            shapes.push(Shape::line(
                std::mem::replace(&mut buf, Vec::with_capacity(TROZO + 1)),
                stroke,
            ));
            buf.push(s); // el siguiente trozo continúa donde terminó este
        }
        if i == ultimo {
            break;
        }
        i = (i + paso).min(ultimo);
    }
    if buf.len() > 1 {
        shapes.push(Shape::line(buf, stroke));
    }
    painter.extend(shapes);
}

/// Dibuja cuadrados (coordenadas del mundo) en `Mesh`es.
/// - `hasta`: solo se dibujan los que ya "nacieron" (`tiempo <= hasta`).
/// - `paleta`: si existe, tiñe cada cuadrado con el color de su hilo.
pub fn cuadrados(
    painter: &egui::Painter,
    camara: &Camara,
    centro: Pos2,
    rect: egui::Rect,
    lista: &[Cuadrado],
    hasta: f32,
    paleta: Option<&[Color32]>,
) {
    let origen = camara.origen(centro);
    let z = camara.zoom;

    let mut lote = Lote::nuevo();
    let mut bordes: Vec<Shape> = Vec::new();

    for c in lista {
        if c.tiempo > hasta {
            continue;
        }
        let lado_px = c.lado * z;
        if lado_px < 0.7 {
            continue;
        }
        let p0 = origen + c.p[0].to_vec2() * z;
        if !rect.expand(lado_px * 2.0).contains(p0) {
            continue;
        }
        let pts = [
            p0,
            origen + c.p[1].to_vec2() * z,
            origen + c.p[2].to_vec2() * z,
            origen + c.p[3].to_vec2() * z,
        ];

        let color = match paleta {
            Some(p) if !p.is_empty() => hilos::mezclar(c.color, p[c.hilo as usize % p.len()], 0.8),
            _ => c.color,
        };

        let i = lote.mesh.vertices.len() as u32;
        for p in pts {
            lote.mesh.colored_vertex(p, color);
        }
        lote.mesh.add_triangle(i, i + 1, i + 2);
        lote.mesh.add_triangle(i, i + 2, i + 3);

        if lado_px > 6.0 && bordes.len() < MAX_BORDES {
            bordes.push(Shape::closed_line(
                pts.to_vec(),
                Stroke::new(0.5_f32, Color32::BLACK),
            ));
        }

        lote.revisar();
        if lote.lleno() {
            break;
        }
    }

    lote.pintar(painter);
    painter.extend(bordes);
}

// ---------- Discos (Romanesco) ----------

const SEGMENTOS: [usize; 4] = [4, 8, 14, 24];
static TABLAS: OnceLock<[Vec<Vec2>; 4]> = OnceLock::new();

fn tablas() -> &'static [Vec<Vec2>; 4] {
    TABLAS.get_or_init(|| {
        SEGMENTOS.map(|n| {
            (0..n)
                .map(|k| {
                    let a = k as f32 / n as f32 * std::f32::consts::TAU;
                    egui::vec2(a.cos(), a.sin())
                })
                .collect()
        })
    })
}

/// Añade un disco relleno al mesh con un número de lados acorde a su tamaño en pantalla.
pub fn disco(mesh: &mut Mesh, centro: Pos2, r: f32, color: Color32) {
    let idx = if r < 2.0 {
        0
    } else if r < 5.0 {
        1
    } else if r < 14.0 {
        2
    } else {
        3
    };
    let unidades = &tablas()[idx];
    let n = unidades.len() as u32;

    let i0 = mesh.vertices.len() as u32;
    mesh.colored_vertex(centro, color);
    for u in unidades {
        mesh.colored_vertex(centro + *u * r, color);
    }
    for k in 0..n {
        mesh.add_triangle(i0, i0 + 1 + k, i0 + 1 + (k + 1) % n);
    }
}
//! Utilidades para visualizar el reparto de trabajo entre "hilos":
//! colores, planificador y panel de progreso por hilo.

use eframe::egui::{self, Align2, Color32, FontId, Painter, Rect};

/// Número de trabajadores que se muestran (hilos del pool de rayon, entre 2 y 12).
pub fn trabajadores() -> usize {
    rayon::current_num_threads().clamp(2, 12)
}

/// Color distinto y estable para cada hilo (ángulo áureo sobre el tono).
pub fn color_hilo(hilo: usize) -> Color32 {
    let h = (0.08 + hilo as f32 * 0.618_034).fract();
    Color32::from(egui::ecolor::Hsva::new(h, 0.70, 0.95, 1.0))
}

pub fn mezclar(a: Color32, b: Color32, t: f32) -> Color32 {
    let t = t.clamp(0.0, 1.0);
    let f = |x: u8, y: u8| (x as f32 + (y as f32 - x as f32) * t) as u8;
    Color32::from_rgb(f(a.r(), b.r()), f(a.g(), b.g()), f(a.b(), b.b()))
}

/// Reparto dinámico: cada tarea va al trabajador con menos carga acumulada
/// (como la cola de trabajo de un pool). Devuelve (trabajador, instante de inicio) por tarea.
pub fn planificar(costes: &[usize], w: usize) -> Vec<(usize, usize)> {
    let mut carga = vec![0usize; w.max(1)];
    costes
        .iter()
        .map(|&c| {
            let i = carga
                .iter()
                .enumerate()
                .min_by_key(|&(_, &l)| l)
                .map(|(i, _)| i)
                .unwrap_or(0);
            let inicio = carga[i];
            carga[i] += c;
            (i, inicio)
        })
        .collect()
}

pub struct FilaHilo {
    pub hilo: usize,
    pub progreso: f32,
}

/// Instante (en unidades de trabajo) en que cada hilo termina cada uno de sus elementos.
pub struct Reparto {
    pub tiempos: Vec<Vec<f32>>,
    pub total: f32,
}

impl Reparto {
    pub fn nuevo(w: usize) -> Self {
        Self { tiempos: vec![Vec::new(); w.max(1)], total: 0.0 }
    }

    /// Hay que registrar los elementos de cada hilo en orden temporal creciente.
    pub fn registrar(&mut self, hilo: usize, t: f32) {
        self.tiempos[hilo].push(t);
        if t > self.total {
            self.total = t;
        }
    }

    /// Solo los hilos que ya nacieron (su primer elemento ya se terminó), en orden de
    /// nacimiento: las barras van apareciendo conforme el fork-join crea hilos.
    pub fn filas(&self, hasta: f32) -> Vec<FilaHilo> {
        let mut filas: Vec<(f32, FilaHilo)> = self
            .tiempos
            .iter()
            .enumerate()
            .filter_map(|(h, v)| {
                let nacimiento = *v.first()?;
                if nacimiento > hasta {
                    return None;
                }
                let hechos = v.partition_point(|&t| t <= hasta);
                Some((
                    nacimiento,
                    FilaHilo { hilo: h, progreso: hechos as f32 / v.len() as f32 },
                ))
            })
            .collect();
        filas.sort_by(|a, b| a.0.total_cmp(&b.0).then(a.1.hilo.cmp(&b.1.hilo)));
        filas.into_iter().map(|(_, f)| f).collect()
    }
}

impl Default for Reparto {
    fn default() -> Self {
        Self::nuevo(1)
    }
}

/// Panel con una barra de progreso por hilo, en la esquina superior izquierda.
pub fn panel(painter: &Painter, rect: Rect, titulo: &str, filas: &[FilaHilo]) {
    if filas.is_empty() {
        return;
    }
    let alto_fila = 14.0_f32;
    let ancho = 176.0_f32;
    let alto = 28.0 + alto_fila * filas.len() as f32;
    let area = Rect::from_min_size(rect.min + egui::vec2(10.0, 10.0), egui::vec2(ancho, alto));

    painter.rect_filled(area, 6.0_f32, Color32::from_black_alpha(160));
    painter.text(
        area.min + egui::vec2(8.0, 13.0),
        Align2::LEFT_CENTER,
        titulo,
        FontId::proportional(13.0),
        Color32::WHITE,
    );

    for (i, f) in filas.iter().enumerate() {
        let y = area.min.y + 28.0 + i as f32 * alto_fila;
        painter.text(
            egui::pos2(area.min.x + 8.0, y + 6.0),
            Align2::LEFT_CENTER,
            format!("H{}", f.hilo),
            FontId::monospace(10.0),
            Color32::LIGHT_GRAY,
        );
        let barra = Rect::from_min_size(egui::pos2(area.min.x + 40.0, y + 2.0), egui::vec2(126.0, 8.0));
        painter.rect_filled(barra, 2.0_f32, Color32::from_gray(55));
        let lleno = Rect::from_min_size(
            barra.min,
            egui::vec2(barra.width() * f.progreso.clamp(0.0, 1.0), 8.0),
        );
        painter.rect_filled(lleno, 2.0_f32, color_hilo(f.hilo));
    }
}

fn menu_principal(ui: &mut egui::Ui, seleccion: &mut Option<Box<dyn Fractal>>) {
    let ancho_ventana = ui.available_width();
    let ancho_boton = (ancho_ventana * 0.6).clamp(400.0, 900.0);
    let alto_boton = 44.0;
    let separacion = 8.0;

    let botones: [(&str, fn() -> Box<dyn Fractal>); 9] = [
        (" Árbol de Pitágoras", || Box::new(ArbolPitagoras::default())),
        (" Árbol de Pitágoras (Paralelo)", || Box::new(ArbolPitagorasParalelo::default())),
        (" Copo de Koch", || Box::new(CopoKoch::default())),
        (" Copo de Koch (Paralelo)", || Box::new(CopoKochParalelo::default())),
        (" Curva del Dragón", || Box::new(Dragon::default())),
        (" Curva del Dragón (Paralelo)", || Box::new(DragonParalelo::default())),
        (" Brócoli Romanesco", || Box::new(Romanesco::default())),
        (" Brócoli Romanesco (Paralelo)", || Box::new(RomanescoParalelo::default())),
        (" Conjunto de Mandelbrot", || Box::new(Mandelbrot::default())),
    ];

    let alto_contenido = botones.len() as f32 * (alto_boton + separacion) + 120.0;
    let espacio_arriba = ((ui.available_height() - alto_contenido) / 2.0).max(10.0);

    ui.vertical_centered(|ui| {
        ui.add_space(espacio_arriba);
        ui.heading(
            egui::RichText::new("Seleccione la animación")
                .size(32.0)
                .strong(),
        );
        ui.add_space(24.0);

        for (texto, constructor) in botones {
            let boton = egui::Button::new(egui::RichText::new(texto).size(20.0))
                .min_size(egui::vec2(ancho_boton, alto_boton))
                .fill(ui.visuals().widgets.inactive.bg_fill);

            if ui.add(boton).clicked() {
                *seleccion = Some(constructor());
            }
            ui.add_space(separacion);
        }
    });
}

fn vista_fractal(ui: &mut egui::Ui, fractal: &mut Box<dyn Fractal>) -> bool {
    let mut volver = false;

    egui::Panel::top("controles").show_inside(ui, |ui| {
        ui.add_space(6.0);
        ui.horizontal(|ui| {
            if ui
                .add(
                    egui::Button::new(egui::RichText::new("← Volver").size(18.0))
                        .min_size(egui::vec2(120.0, 40.0)),
                )
                .clicked()
            {
                volver = true;
            }
            ui.separator();
            ui.heading(egui::RichText::new(fractal.nombre()).size(24.0));
        });
        ui.separator();
        fractal.ui_controles(ui);
        ui.add_space(6.0);
    });

    egui::CentralPanel::default().show_inside(ui, |ui| {
        let rect = ui.available_rect_before_wrap();
        fractal.dibujar(ui, rect);
    });

    volver
}

impl eframe::App for MyPixelArt {
    fn ui(&mut self, ui: &mut egui::Ui, _frame: &mut eframe::Frame) {
        match &mut self.fractal {
            None => menu_principal(ui, &mut self.fractal),
            Some(fractal) => {
                let volver = vista_fractal(ui, fractal);
                if volver {
                    self.fractal = None;
                }
            }
        }
    }
}

/// Fork-join: mientras al subárbol le queden varios trabajadores, se bifurca con
/// `rayon::join` y los trabajadores se reparten entre las dos ramas (mitad y mitad).
/// Cuando queda uno solo, ese hilo termina el subárbol de forma secuencial.
/// `ini..ini+n` son los trabajadores disponibles; el nodo lo atiende `ini`.
fn generar_par(
    base: Pos2,
    lado: f32,
    angulo: f32,
    nivel: u32,
    prof_max: u32,
    ini: usize,
    n: usize,
) -> Vec<Cuadrado> {
    if nivel >= prof_max || lado < LADO_MIN {
        return Vec::new();
    }
    if n <= 1 {
        let mut v = Vec::new();
        generar_rec(&mut v, base, lado, angulo, nivel, prof_max);
        for c in &mut v {
            c.hilo = ini as u8;
        }
        return v;
    }

    let mut nodo = crear_nodo(base, lado, angulo, nivel, prof_max);
    nodo.cuadrado.hilo = ini as u8;
    let l = lado * R;
    let (base_izq, base_der) = (nodo.base_izq, nodo.base_der);
    let n_izq = (n + 1) / 2;
    let n_der = n - n_izq;

    let (izq, der) = rayon::join(
        || generar_par(base_izq, l, angulo + A45, nivel + 1, prof_max, ini, n_izq),
        || generar_par(base_der, l, angulo - A45, nivel + 1, prof_max, ini + n_izq, n_der),
    );

    let mut v = Vec::with_capacity(1 + izq.len() + der.len());
    v.push(nodo.cuadrado);
    v.extend(izq);
    v.extend(der);
    v
}

impl ArbolPitagorasParalelo {
    fn generar(&mut self) {
        let w = hilos::trabajadores();
        let v = generar_par(
            base_raiz(),
            LADO_RAIZ,
            ANGULO_INICIAL,
            0,
            self.profundidad,
            0,
            w,
        );
        self.trabajadores = w;
        self.geom = Geometria::nueva(v, w);
        self.firma = Some(self.profundidad);
    }
}

/// Lado del cuadrado raíz (unidades del mundo).
pub const LADO_RAIZ: f32 = 140.0;
/// No se generan cuadrados más pequeños que esto (en unidades del mundo).
pub const LADO_MIN: f32 = 0.05;
pub const A45: f32 = FRAC_PI_4;
/// cos(45°) = sin(45°): ambos hijos miden lado * R.
pub const R: f32 = FRAC_1_SQRT_2;
/// Cada cuadrado "crece" hacia su perpendicular (-sen, cos). Con -PI esa dirección
/// apunta hacia arriba en pantalla (Y negativo), así el tronco sube.
pub const ANGULO_INICIAL: f32 = -PI;

pub struct Nodo {
    pub cuadrado: Cuadrado,
    pub base_izq: Pos2,
    pub base_der: Pos2,
}

/// Calcula el cuadrado de un nodo y las bases de sus dos hijos.
pub fn crear_nodo(base: Pos2, lado: f32, angulo: f32, nivel: u32, prof_max: u32) -> Nodo {
    let (sin, cos) = angulo.sin_cos();
    let v1 = base + egui::vec2(cos * lado, sin * lado);
    let v2 = v1 + egui::vec2(-sin * lado, cos * lado);
    let v3 = base + egui::vec2(-sin * lado, cos * lado);
    // Ápice del triángulo rectángulo sobre el lado superior: ahí nace el hijo derecho.
    // (Es el primer vértice del hijo izquierdo: v3 + dirección(ángulo + 45°) * lado * cos 45°.)
    let (sin_h, cos_h) = (angulo + A45).sin_cos();
    let apice = v3 + egui::vec2(cos_h * lado * R, sin_h * lado * R);

    // t = 0 en la raíz, t = 1 en las hojas. Café -> verde claro.
    let t = if prof_max > 1 {
        (nivel as f32 / (prof_max - 1) as f32).clamp(0.0, 1.0)
    } else {
        0.0
    };
    let color = Color32::from_rgb(
        (139.0 + (100.0 - 139.0) * t) as u8,
        (90.0 + (200.0 - 90.0) * t) as u8,
        (43.0 + (80.0 - 43.0) * t) as u8,
    );

    Nodo {
        cuadrado: Cuadrado {
            p: [base, v1, v2, v3],
            lado,
            color,
            nivel: nivel.min(255) as u8,
            hilo: 0,
            tiempo: 0.0,
        },
        base_izq: v3,
        base_der: apice,
    }
}

/// Recursión secuencial en preorden (DFS). Escribe en coordenadas del mundo.
pub fn generar_rec(
    out: &mut Vec<Cuadrado>,
    base: Pos2,
    lado: f32,
    angulo: f32,
    nivel: u32,
    prof_max: u32,
) {
    if nivel >= prof_max || lado < LADO_MIN {
        return;
    }
    let n = crear_nodo(base, lado, angulo, nivel, prof_max);
    out.push(n.cuadrado);
    let l = lado * R;
    generar_rec(out, n.base_izq, l, angulo + A45, nivel + 1, prof_max);
    generar_rec(out, n.base_der, l, angulo - A45, nivel + 1, prof_max);
}

/// Base del cuadrado raíz. La posición exacta da igual: `Geometria::nueva` centra el árbol.
pub fn base_raiz() -> Pos2 {
    egui::pos2(LADO_RAIZ / 2.0, 0.0)
}

/// Cuadrados en preorden + el instante en que cada hilo termina cada uno.
pub struct Geometria {
    pub cuadrados: Vec<Cuadrado>,
    pub reparto: Reparto,
}

impl Default for Geometria {
    fn default() -> Self {
        Self { cuadrados: Vec::new(), reparto: Reparto::default() }
    }
}

impl Geometria {
    pub fn nueva(mut cuadrados: Vec<Cuadrado>, trabajadores: usize) -> Self {
        centrar(&mut cuadrados);
        let reparto = asignar_tiempos(&mut cuadrados, trabajadores);
        Self { cuadrados, reparto }
    }
}

/// Centra el árbol en el origen usando la caja que envuelve todos sus vértices.
fn centrar(cuadrados: &mut [Cuadrado]) {
    if cuadrados.is_empty() {
        return;
    }
    let (mut min_x, mut min_y, mut max_x, mut max_y) = (f32::MAX, f32::MAX, f32::MIN, f32::MIN);
    for c in cuadrados.iter() {
        for p in &c.p {
            min_x = min_x.min(p.x);
            max_x = max_x.max(p.x);
            min_y = min_y.min(p.y);
            max_y = max_y.max(p.y);
        }
    }
    let d = egui::vec2(-(min_x + max_x) / 2.0, -(min_y + max_y) / 2.0);
    for c in cuadrados.iter_mut() {
        for p in &mut c.p {
            *p += d;
        }
    }
}

/// Simula la ejecución: cada hilo procesa sus nodos uno tras otro (1 unidad de trabajo
/// por nodo) y un nodo no puede empezar antes de que termine su padre.
/// Con 1 hilo el instante de cada nodo es simplemente su posición en el preorden.
pub fn asignar_tiempos(cuadrados: &mut [Cuadrado], trabajadores: usize) -> Reparto {
    let w = trabajadores.max(1);
    let mut libre = vec![0.0f32; w];
    let mut pila = [0.0f32; 256]; // instante del último ancestro visto en cada nivel
    let mut reparto = Reparto::nuevo(w);

    for c in cuadrados.iter_mut() {
        let h = (c.hilo as usize).min(w - 1);
        let nivel = c.nivel as usize;
        let t_padre = if nivel > 0 { pila[nivel - 1] } else { 0.0 };
        let t = libre[h].max(t_padre) + 1.0;
        libre[h] = t;
        pila[nivel] = t;
        c.tiempo = t;
        reparto.registrar(h, t);
    }
    reparto
}

/// Dibujo común a las dos versiones del árbol (la cámara ya se procesó fuera).
pub fn dibujar_arbol(
    ui: &mut egui::Ui,
    rect: egui::Rect,
    camara: &Camara,
    anim: &mut Animacion,
    geom: &Geometria,
    por_hilo: bool,
    titulo: &str,
) {
    let centro = rect.center();
    let total = geom.reparto.total.ceil() as usize;
    let n = anim.progreso(ui, total);
    let hasta = if anim.animando { n as f32 } else { f32::INFINITY };

    let paleta: Vec<Color32> = (0..geom.reparto.tiempos.len()).map(hilos::color_hilo).collect();
    let painter = ui.painter_at(rect);
    dibujo::cuadrados(
        &painter,
        camara,
        centro,
        rect,
        &geom.cuadrados,
        hasta,
        if por_hilo { Some(paleta.as_slice()) } else { None },
    );
    hilos::panel(&painter, rect, titulo, &geom.reparto.filas(hasta));
}

/// Por debajo de este nivel restante se calcula en secuencial
/// (paralelizar tareas diminutas cuesta más que el trabajo útil).
const UMBRAL_PARALELO: u32 = 6;

/// Rellena `out` (longitud exacta 4^prof) con el punto de inicio de cada segmento hoja.
/// Sin Vec intermedios: cada tarea escribe directo en su porción del slice.
fn rellenar(a: Pos2, b: Pos2, prof: u32, out: &mut [Pos2]) {
    if prof == 0 {
        out[0] = a;
        return;
    }
    let (p1, p2, p3) = koch_puntos(a, b);
    let tareas = [(a, p1), (p1, p2), (p2, p3), (p3, b)];
    let q = out.len() / 4;

    if prof > UMBRAL_PARALELO {
        out.par_chunks_mut(q)
            .zip(tareas.as_slice().par_iter())
            .for_each(|(trozo, &(s, e))| rellenar(s, e, prof - 1, trozo));
    } else {
        for (trozo, &(s, e)) in out.chunks_mut(q).zip(tareas.iter()) {
            rellenar(s, e, prof - 1, trozo);
        }
    }
}

/// Dirección del segmento `k` en forma cerrada: el rumbo (en múltiplos de 90°)
/// es el número de unos del código Gray de k. Por eso cada bloque puede calcularse
/// sin conocer los anteriores.
fn direccion(k: usize) -> Vec2 {
    match (k ^ (k >> 1)).count_ones() % 4 {
        0 => egui::vec2(0.0, -PASO_MUNDO),
        1 => egui::vec2(PASO_MUNDO, 0.0),
        2 => egui::vec2(0.0, PASO_MUNDO),
        _ => egui::vec2(-PASO_MUNDO, 0.0),
    }
}

impl DragonParalelo {
    /// Suma de prefijos (scan) paralela en 3 fases:
    ///  1. (paralela)   cada bloque suma sus desplazamientos
    ///  2. (secuencial) prefijos sobre las pocas sumas de bloque
    ///  3. (paralela)   cada bloque escribe sus puntos partiendo de su prefijo
    fn generar(&mut self) {
        let w = hilos::trabajadores();
        let n = 1usize << self.profundidad;
        let tam = ((n + w - 1) / w).max(1);
        let origen = egui::vec2(-100.0, 50.0);

        let sumas: Vec<Vec2> = (0..w)
            .into_par_iter()
            .map(|b| {
                let ini = (b * tam).min(n);
                let fin = ((b + 1) * tam).min(n);
                let mut s = Vec2::ZERO;
                for k in ini..fin {
                    s += direccion(k);
                }
                s
            })
            .collect();

        let mut inicios: Vec<Vec2> = Vec::with_capacity(w);
        let mut acc = origen;
        for s in &sumas {
            inicios.push(acc);
            acc += *s;
        }

        let mut puntos = vec![Pos2::ZERO; n + 1];
        puntos[0] = origen.to_pos2();
        puntos[1..]
            .par_chunks_mut(tam)
            .enumerate()
            .for_each(|(b, trozo)| {
                let ini = b * tam;
                let mut p = inicios[b].to_pos2();
                for (j, slot) in trozo.iter_mut().enumerate() {
                    p += direccion(ini + j);
                    *slot = p;
                }
            });

        self.trabajadores = w;
        self.curva = Curva::bloques(puntos, w);
        self.firma = Some(self.profundidad);
    }
}

        let paso = ((1.0 / largo_px).ceil() as usize).clamp(1, total.max(1));

        );
        let titulo = format!("Paralelo: {} hilos", self.trabajadores);
        hilos::panel(&painter, rect, &titulo, &self.curva.filas(n));

        let mut puntos: Vec<Pos2> = Vec::with_capacity(n + 1);

        let mut p = egui::pos2(-100.0, 50.0);
        let mut dir = egui::vec2(0.0, -PASO_MUNDO);

        p += dir;

                egui::vec2(-dir.y, dir.x) // +90°

            p += dir;
            puntos.push(p);

//! Conjunto de Mandelbrot que se calcula "en vivo": cada frame los trabajadores
//! resuelven unas cuantas filas reales y la imagen va apareciendo.
//! - Secuencial: 1 trabajador recorre la imagen de arriba abajo.
//! - Paralelo: varios trabajadores (rayon) avanzan a la vez, ya sea en bandas contiguas
//!   o con filas intercaladas (así se ve el efecto del balanceo de carga: las filas del
//!   centro cuestan mucho más que las de los bordes).

use super::Fractal;
use crate::hilos::{self, FilaHilo};
use eframe::egui::{self, Color32, ColorImage, Rect, Stroke, TextureHandle, TextureOptions};
use rayon::prelude::*;

const TAM_PALETA: usize = 1024;
const LN2: f64 = std::f64::consts::LN_2;

/// `escala` = unidades del plano complejo por píxel de pantalla.
#[derive(Clone, Copy, PartialEq)]
struct Vista {
    cx: f64,
    cy: f64,
    escala: f64,
}

/// Todo lo que, si cambia, obliga a recalcular la imagen desde cero.
#[derive(Clone, Copy, PartialEq)]
struct Firma {
    w: usize,
    h: usize,
    vista: Vista,
    max_iter: u32,
    bloque: u32,
    paralelo: bool,
    intercalado: bool,
}

#[derive(Clone, Copy)]
struct Params {
    w: usize,
    h: usize,
    cx: f64,
    cy: f64,
    /// Unidades complejas por píxel de la imagen (escala * tamaño de bloque).
    paso: f64,
    max_iter: u32,
}

struct Trabajador {
    filas: Vec<usize>,
    cursor: usize,
}

fn crear_paleta() -> Vec<[u8; 3]> {
    (0..TAM_PALETA)
        .map(|i| {
            let t = i as f32 / TAM_PALETA as f32;
            let c = |fase: f32| {
                ((0.5 + 0.5 * (std::f32::consts::TAU * (t + fase)).cos()) * 255.0) as u8
            };
            [c(0.0), c(0.33), c(0.67)]
        })
        .collect()
}

/// Iteraciones "suaves" hasta escapar, o -1.0 si el punto no escapa (pertenece al conjunto).
fn iterar(cr: f64, ci: f64, max_iter: u32) -> f64 {
    // Atajos: cardioide principal y bulbo de periodo 2 están dentro del conjunto.
    let q = (cr - 0.25) * (cr - 0.25) + ci * ci;
    if q * (q + (cr - 0.25)) <= 0.25 * ci * ci {
        return -1.0;
    }
    if (cr + 1.0) * (cr + 1.0) + ci * ci <= 0.0625 {
        return -1.0;
    }

    let (mut zr, mut zi) = (0.0f64, 0.0f64);
    let (mut zr2, mut zi2) = (0.0f64, 0.0f64);
    let mut i = 0u32;
    // Radio de escape 16 (|z|² > 256): da un degradado de color más suave.
    while i < max_iter && zr2 + zi2 <= 256.0 {
        zi = 2.0 * zr * zi + ci;
        zr = zr2 - zi2 + cr;
        zr2 = zr * zr;
        zi2 = zi * zi;
        i += 1;
    }
    if i >= max_iter {
        -1.0
    } else {
        let log_zn = (zr2 + zi2).ln() * 0.5;
        let nu = log_zn.ln() / LN2;
        i as f64 + 1.0 - nu
    }
}

fn calcular_fila(y: usize, p: &Params, paleta: &[[u8; 3]]) -> Vec<u8> {
    let mut out = Vec::with_capacity(p.w * 4);
    let ci = p.cy + (y as f64 + 0.5 - p.h as f64 / 2.0) * p.paso;
    for x in 0..p.w {
        let cr = p.cx + (x as f64 + 0.5 - p.w as f64 / 2.0) * p.paso;
        let mu = iterar(cr, ci, p.max_iter);
        let rgb = if mu < 0.0 {
            [0, 0, 0]
        } else {
            paleta[((mu / 32.0).fract() * TAM_PALETA as f64) as usize % TAM_PALETA]
        };
        out.extend_from_slice(&[rgb[0], rgb[1], rgb[2], 255]);
    }
    out
}

fn avanzar_trabajador(
    t: &mut Trabajador,
    filas_por_frame: usize,
    p: &Params,
    paleta: &[[u8; 3]],
) -> Vec<(usize, Vec<u8>)> {
    let hasta = (t.cursor + filas_por_frame).min(t.filas.len());
    let hechas = t.filas[t.cursor..hasta]
        .iter()
        .map(|&y| (y, calcular_fila(y, p, paleta)))
        .collect();
    t.cursor = hasta;
    hechas
}

pub struct Mandelbrot {
    max_iter: u32,
    bloque: u32,
    filas_por_frame: usize,
    paralelo: bool,
    intercalado: bool,

    vista: Option<Vista>,
    firma: Option<Firma>,
    trabajadores: Vec<Trabajador>,
    w: usize,
    h: usize,
    rgba: Vec<u8>,
    textura: Option<TextureHandle>,
    paleta: Vec<[u8; 3]>,
    sucio: bool,
    acabado: bool,
}

impl Default for Mandelbrot {
    fn default() -> Self {
        Self {
            max_iter: 300,
            bloque: 2,
            filas_por_frame: 1,
            paralelo: false,
            intercalado: false,
            vista: None,
            firma: None,
            trabajadores: Vec::new(),
            w: 1,
            h: 1,
            rgba: Vec::new(),
            textura: None,
            paleta: crear_paleta(),
            sucio: false,
            acabado: false,
        }
    }
}

impl Mandelbrot {
    fn reiniciar(&mut self, firma: Firma) {
        self.w = firma.w;
        self.h = firma.h;
        self.rgba = vec![0u8; self.w * self.h * 4];
        for px in self.rgba.chunks_exact_mut(4) {
            px.copy_from_slice(&[18, 18, 22, 255]);
        }

        let nw = if firma.paralelo { hilos::trabajadores() } else { 1 };
        let h = self.h;
        self.trabajadores = (0..nw)
            .map(|k| {
                let filas: Vec<usize> = if nw == 1 {
                    (0..h).collect()
                } else if firma.intercalado {
                    (k..h).step_by(nw).collect()
                } else {
                    let tam = (h + nw - 1) / nw;
                    ((k * tam).min(h)..((k + 1) * tam).min(h)).collect()
                };
                Trabajador { filas, cursor: 0 }
            })
            .collect();

        self.textura = None; // se recrea con el nuevo tamaño
        self.sucio = true;
        self.acabado = false;
        self.firma = Some(firma);
    }

    /// Cada trabajador resuelve hasta `filas_por_frame` filas reales.
    /// En paralelo lo hacen a la vez (rayon); en secuencial es un único recorrido.
    fn avanzar(&mut self) {
        if self.acabado {
            return;
        }
        let Some(firma) = self.firma else { return };
        let p = Params {
            w: self.w,
            h: self.h,
            cx: firma.vista.cx,
            cy: firma.vista.cy,
            paso: firma.vista.escala * firma.bloque as f64,
            max_iter: firma.max_iter,
        };
        let l = self.filas_por_frame.max(1);
        let paleta = &self.paleta;

        let resultados: Vec<Vec<(usize, Vec<u8>)>> = if firma.paralelo {
            self.trabajadores
                .par_iter_mut()
                .map(|t| avanzar_trabajador(t, l, &p, paleta))
                .collect()
        } else {
            self.trabajadores
                .iter_mut()
                .map(|t| avanzar_trabajador(t, l, &p, paleta))
                .collect()
        };

        for filas in resultados {
            for (y, datos) in filas {
                let ini = y * self.w * 4;
                self.rgba[ini..ini + datos.len()].copy_from_slice(&datos);
                self.sucio = true;
            }
        }
        self.acabado = self
            .trabajadores
            .iter()
            .all(|t| t.cursor >= t.filas.len());
    }
}

            ui.label("Filas por frame (por hilo):");
            ui.add(egui::Slider::new(&mut self.filas_por_frame, 1..=64));
            ui.checkbox(&mut self.paralelo, "Paralelo (rayon)");
            ui.add_enabled(

                egui::Checkbox::new(&mut self.intercalado, "Filas intercaladas"),
            );
            if ui.button("Repetir animación").clicked() {
                self.firma = None;
            }
            if ui.button("Reset zoom").clicked() {
                self.vista = None;
            }
        });
    }

    fn dibujar(&mut self, ui: &mut egui::Ui, rect: egui::Rect) {
        let bloque = self.bloque.max(1) as usize;
        let w = ((rect.width() as usize) / bloque).max(1);
        let h = ((rect.height() as usize) / bloque).max(1);
        let dest = Rect::from_min_size(
            rect.min,
            egui::vec2((w * bloque) as f32, (h * bloque) as f32),
        );

        // Interacción: arrastrar mueve la vista, la rueda hace zoom hacia el cursor.
        let resp = ui.allocate_rect(rect, egui::Sense::drag());
        let v = self.vista.get_or_insert_with(|| Vista {
            cx: -0.5,
            cy: 0.0,
            escala: 2.8 / rect.height().max(1.0) as f64,
        });
        if resp.dragged() {
            let d = resp.drag_delta();
            v.cx -= d.x as f64 * v.escala;
            v.cy -= d.y as f64 * v.escala;
        }
        if resp.hovered() {
            let scroll = ui.input(|i| i.smooth_scroll_delta.y);
            if scroll != 0.0 {
                if let Some(cursor) = resp.hover_pos() {
                    let factor = (scroll as f64 * 0.002).exp(); // > 1: acercar
                    let off = cursor - dest.center();
                    let nueva = (v.escala / factor).clamp(1e-13, 0.05);
                    v.cx += off.x as f64 * (v.escala - nueva);
                    v.cy += off.y as f64 * (v.escala - nueva);
                    v.escala = nueva;
                }
            }
        }
        let vista = *v;

        // Si cambió algo (vista, tamaño, parámetros) se vuelve a empezar.
        let firma = Firma {
            w,
            h,
            vista,
            max_iter: self.max_iter,
            bloque: self.bloque.max(1),
            paralelo: self.paralelo,
            intercalado: self.intercalado,
        };
        if self.firma != Some(firma) {
            self.reiniciar(firma);
        }

        self.avanzar();

        if self.sucio {
            let img = ColorImage::from_rgba_unmultiplied([self.w, self.h], &self.rgba);
            if let Some(t) = self.textura.as_mut() {
                t.set(img, TextureOptions::NEAREST);
            } else {
                self.textura = Some(ui.ctx().load_texture("mandelbrot", img, TextureOptions::NEAREST));
            }
            self.sucio = false;
        }
        if !self.acabado {
            ui.ctx().request_repaint();
        }

        let painter = ui.painter_at(rect);
        if let Some(t) = &self.textura {
            painter.image(
                t.id(),
                dest,
                Rect::from_min_max(egui::pos2(0.0, 0.0), egui::pos2(1.0, 1.0)),
                Color32::WHITE,
            );
        }

        // Una línea de color por trabajador marca la fila que está resolviendo.
        if !self.acabado {
            for (k, t) in self.trabajadores.iter().enumerate() {
                if t.cursor < t.filas.len() {
                    let y = dest.min.y + (t.filas[t.cursor] as f32 + 0.5) * bloque as f32;
                    painter.line_segment(
                        [egui::pos2(dest.min.x, y), egui::pos2(dest.max.x, y)],
                        Stroke::new(2.0_f32, hilos::color_hilo(k)),
                    );
                }
            }
        }

        let filas: Vec<FilaHilo> = self
            .trabajadores
            .iter()
            .enumerate()
            .map(|(k, t)| FilaHilo {
                hilo: k,
                progreso: if t.filas.is_empty() {
                    1.0
                } else {
                    t.cursor as f32 / t.filas.len() as f32
                },
            })
            .collect();
        let titulo = if self.trabajadores.len() > 1 {
            format!("Paralelo: {} hilos", self.trabajadores.len())
        } else {
            "Secuencial: 1 hilo".to_string()
        };
        hilos::panel(&painter, rect, &titulo, &filas);
    }
}

// ---------- Utilidades compartidas por los dos copos de Koch ----------

pub const LADO_TRIANGULO_KOCH: f32 = 200.0;

/// Son constantes para no llamar a `sin_cos` millones de veces en profundidades altas.
const COS_KOCH: f32 = 0.5;
const SEN_KOCH: f32 = -0.866_025_4;

/// Triángulo equilátero centrado en el origen (coordenadas del mundo).
pub fn triangulo_koch() -> [egui::Pos2; 3] {
    let r = LADO_TRIANGULO_KOCH / 3.0_f32.sqrt();
    let mitad = LADO_TRIANGULO_KOCH / 2.0;
    [
        egui::pos2(0.0, -r),
        egui::pos2(mitad, r * 0.5),
        egui::pos2(-mitad, r * 0.5),
    ]
}

/// Dado un segmento a-b devuelve p1, p2 (la punta) y p3 de la curva de Koch.
pub fn koch_puntos(a: egui::Pos2, b: egui::Pos2) -> (egui::Pos2, egui::Pos2, egui::Pos2) {
    let t = (b - a) / 3.0;
    let p1 = a + t;
    let p3 = a + t * 2.0;
    let p2 = p1 + egui::vec2(t.x * COS_KOCH - t.y * SEN_KOCH, t.x * SEN_KOCH + t.y * COS_KOCH);
    (p1, p2, p3)
}

    /// Cada "cabeza" del primer anillo es una tarea independiente (ella y todo su
    /// subárbol de cabezas más pequeñas). Se calculan en paralelo con rayon y luego se
    /// reparten entre los trabajadores con una cola dinámica: cada tarea va al hilo
    /// menos cargado, así que las cabezas grandes y las pequeñas se compensan.
    fn generar(&mut self) {
        let w = hilos::trabajadores();
        let prof = self.profundidad;

        let raices = hijos(egui::pos2(0.0, BASE_Y), RADIO_BASE);
        let subs: Vec<Vec<Bulto>> = raices
            .into_par_iter()
            .map(|b| {
                let mut v = Vec::new();
                expandir(&mut v, b, prof);
                v
            })
            .collect();

        let costes: Vec<usize> = subs.iter().map(|s| s.len()).collect();
        let plan = hilos::planificar(&costes, w);

        let total: usize = costes.iter().sum();
        let mut todos = Vec::with_capacity(total);
        let mut reparto = Reparto::nuevo(w);
        for (mut sub, &(hilo, inicio)) in subs.into_iter().zip(plan.iter()) {
            for (j, b) in sub.iter_mut().enumerate() {
                b.hilo = hilo as u8;
                b.tiempo = (inicio + j + 1) as f32;
                reparto.registrar(hilo, b.tiempo);
            }
            todos.extend(sub);
        }

        self.trabajadores = w;
        self.bultos = todos;
        self.reparto = reparto;
        self.firma = Some(prof);
    }
}

    }

use crate::animacion::Animacion;

/// Tamaño del romanesco en unidades del mundo (pensado para una ventana de ~1000 px de alto).
pub const RADIO_BASE: f32 = 370.0;
pub const BASE_Y: f32 = 264.0;

const ANGULO_DORADO: f32 = 2.399_963_2;
const NIVELES: usize = 8;
const FIB: [u32; NIVELES] = [5, 8, 13, 21, 21, 13, 8, 5];
const RADIO_MINIMO: f32 = 1.5;
/// 1.0 = niveles tangentes, <1.0 = se solapan, >1.0 = separados.
const SOLAPE_Y: f32 = 0.7;

#[derive(Clone, Copy)]
pub struct Bulto {
    pub pos: Pos2,
    pub radio: f32,
    pub color: Color32,
    pub hilo: u8,
    /// Instante (en unidades de trabajo) en que su hilo lo termina.
    pub tiempo: f32,
}

fn anillo(nivel: usize, radio_base: f32) -> (f32, f32, u32) {
    let t = nivel as f32 / (NIVELES as f32 - 1.0).max(1.0);
    let radio_nivel = radio_base * (1.0 - t);

    let fib = FIB[nivel.min(FIB.len() - 1)];
    let cap = (radio_nivel * std::f32::consts::PI / RADIO_MINIMO) as u32;
    let num_bultos = fib.min(cap).max(1);

    let radio_bulto = (std::f32::consts::PI * radio_nivel / num_bultos as f32) * 1.05;
    (radio_nivel, radio_bulto, num_bultos)
}

fn alturas_niveles(radio_base: f32) -> [f32; NIVELES] {
    let mut ys = [0.0f32; NIVELES];
    let mut cum = 0.0f32;
    for nivel in 0..NIVELES {
        ys[nivel] = cum;
        let (_, radio_bulto, _) = anillo(nivel, radio_base);
        cum += radio_bulto * SOLAPE_Y;
    }
    ys
}

/// Los bultos de UNA torre (sus 8 anillos), sin recursión.
pub fn hijos(centro: Pos2, radio_base: f32) -> Vec<Bulto> {
    let mut out = Vec::new();
    if radio_base < RADIO_MINIMO {
        return out;
    }
    let ys = alturas_niveles(radio_base);

    for nivel in 0..NIVELES {
        let t = nivel as f32 / (NIVELES as f32 - 1.0).max(1.0);
        let y = centro.y - ys[nivel];
        let (radio_nivel, radio_bulto, num_bultos) = anillo(nivel, radio_base);
        if radio_bulto < RADIO_MINIMO {
            continue;
        }

        let offset_nivel = nivel as f32 * ANGULO_DORADO * 0.5;
        let color = Color32::from_rgb(
            (40.0 + 100.0 * t) as u8,
            (110.0 + 100.0 * t) as u8,
            (30.0 + 60.0 * t) as u8,
        );

        for i in 0..num_bultos {
            let angulo = i as f32 * ANGULO_DORADO + offset_nivel;
            let offset_x = angulo.cos() * radio_nivel;
            let offset_z = angulo.sin() * radio_nivel * 0.35;
            out.push(Bulto {
                pos: egui::pos2(centro.x + offset_x, y + offset_z),
                radio: radio_bulto,
                color,
                hilo: 0,
                tiempo: 0.0,
            });
        }
    }
    out
}

/// Preorden: el bulto y, recursivamente, su propia torre de bultos más pequeños.
pub fn expandir(out: &mut Vec<Bulto>, b: Bulto, profundidad: u32) {
    out.push(b);
    if profundidad > 1 && b.radio > 3.0 {
        for h in hijos(b.pos, b.radio * 0.85) {
            expandir(out, h, profundidad - 1);
        }
    }
}

/// Dibujo común a las dos versiones (la cámara ya se procesó fuera).
pub fn dibujar_bultos(
    ui: &mut egui::Ui,
    rect: egui::Rect,
    camara: &Camara,
    anim: &mut Animacion,
    bultos: &[Bulto],
    reparto: &Reparto,
    por_hilo: bool,
    titulo: &str,
) {
    let centro = rect.center();
    let total = reparto.total.ceil() as usize;
    let n = anim.progreso(ui, total);
    let hasta = if anim.animando { n as f32 } else { f32::INFINITY };

    let origen = camara.origen(centro);
    let z = camara.zoom;
    let paleta: Vec<Color32> = (0..reparto.tiempos.len()).map(hilos::color_hilo).collect();
    let borde = Color32::from_rgb(20, 60, 20);

    let mut lote = Lote::nuevo();
    for b in bultos {
        if b.tiempo > hasta {
            continue;
        }
        let r = b.radio * z;
        if r < 0.5 {
            continue; // LOD: invisible
        }
        let pos = origen + b.pos.to_vec2() * z;
        if !rect.expand(r).contains(pos) {
            continue; // culling
        }

        let color = if por_hilo {
            hilos::mezclar(b.color, paleta[b.hilo as usize % paleta.len()], 0.7)
        } else {
            b.color
        };

        if r > 5.0 {
            dibujo::disco(&mut lote.mesh, pos, r, borde);
            dibujo::disco(&mut lote.mesh, pos, r - 0.9, color);
        } else {
            dibujo::disco(&mut lote.mesh, pos, r, color);
        }

        lote.revisar();
        if lote.lleno() {
            break;
        }
    }

    let painter = ui.painter_at(rect);
    lote.pintar(&painter);
    hilos::panel(&painter, rect, titulo, &reparto.filas(hasta));
}

pub struct Romanesco {
    pub profundidad: u32,
    pub camara: Camara,
    anim: Animacion,
    bultos: Vec<Bulto>,
    reparto: Reparto,
    firma: Option<u32>,
}

impl Default for Romanesco {
    fn default() -> Self {
        Self {
            profundidad: 2,
            camara: Camara::default(),
            anim: Animacion::new(400.0),
            bultos: Vec::new(),
            reparto: Reparto::default(),
            firma: None,
        }
    }
}

impl Romanesco {
    fn generar(&mut self) {
        let mut v = Vec::new();
        for b in hijos(egui::pos2(0.0, BASE_Y), RADIO_BASE) {
            expandir(&mut v, b, self.profundidad);
        }

        // Un solo hilo: el instante de cada bulto es su posición en el preorden.
        let mut reparto = Reparto::nuevo(1);
        for (i, b) in v.iter_mut().enumerate() {
            b.hilo = 0;
            b.tiempo = (i + 1) as f32;
            reparto.registrar(0, b.tiempo);
        }

        self.bultos = v;
        self.reparto = reparto;
        self.firma = Some(self.profundidad);
    }
}

impl Fractal for Romanesco {
    fn nombre(&self) -> &'static str {
        "Brócoli Romanesco"
    }

    fn ui_controles(&mut self, ui: &mut egui::Ui) {

            ui.label("Profundidad:");
            if ui.add(egui::Slider::new(&mut self.profundidad, 1..=3)).changed() {
                self.anim.reiniciar();
            }
            ui.label("Velocidad:");
            ui.add(egui::Slider::new(&mut self.anim.velocidad, 10.0..=200000.0).logarithmic(true));
            if ui.button("Repetir animación").clicked() {
                self.anim.reiniciar();
            }
            if ui.button("Reset zoom").clicked() {
                self.camara.reset();
            }
        });
    }

    fn dibujar(&mut self, ui: &mut egui::Ui, rect: egui::Rect) {
        self.camara.interactuar(ui, rect);

        if self.firma != Some(self.profundidad) {
            self.generar();
        }

        dibujar_bultos(
            ui,
            rect,
            &self.camara,
            &mut self.anim,
            &self.bultos,
            &self.reparto,
            false,
            "Secuencial: 1 hilo",
        );
    }
}
